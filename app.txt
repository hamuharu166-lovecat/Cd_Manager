import streamlit as st
import sqlite3
import datetime as dt

DB_NAME = "music.db"

# -----------------------------
# DB 初期化
# -----------------------------
def init_db():
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()

    # アルバム
    c.execute("""
        CREATE TABLE IF NOT EXISTS albums (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            catalog_no TEXT,
            title TEXT NOT NULL,
            artist TEXT,
            year INTEGER,
            release_date TEXT,
            omnibus_flag BOOLEAN DEFAULT FALSE,
            album_type TEXT,
            note TEXT
        )
    """)

    # 曲
    c.execute("""
        CREATE TABLE IF NOT EXISTS tracks (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            album_id INTEGER,
            track_no INTEGER,
            title TEXT NOT NULL,
            artist TEXT,
            composer TEXT,
            arranger TEXT,
            lyricist TEXT,
            duration TEXT,
            note TEXT,
            FOREIGN KEY(album_id) REFERENCES albums(id)
        )
    """)

    # ミュージシャン
    c.execute("""
        CREATE TABLE IF NOT EXISTS musicians (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            note TEXT
        )
    """)

    # アーティスト
    c.execute("""
        CREATE TABLE IF NOT EXISTS artists (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL UNIQUE,
            note TEXT
        )
    """)

    # 曲 × ミュージシャン × 楽器
    c.execute("""
        CREATE TABLE IF NOT EXISTS track_musicians (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            track_id INTEGER,
            musician_id INTEGER,
            instrument TEXT,
            note TEXT,
            FOREIGN KEY(track_id) REFERENCES tracks(id),
            FOREIGN KEY(musician_id) REFERENCES musicians(id)
        )
    """)

    # アルバム × 演奏者 × 楽器（新規）
    c.execute("""
        CREATE TABLE IF NOT EXISTS album_musicians (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            album_id INTEGER,
            musician_id INTEGER,
            instrument TEXT,
            note TEXT,
            FOREIGN KEY(album_id) REFERENCES albums(id),
            FOREIGN KEY(musician_id) REFERENCES musicians(id)
        )
    """)

    conn.commit()
    conn.close()

# -----------------------------
# DB 操作（アルバム）
# -----------------------------
def add_album(catalog_no, title, artist, year, release_date, album_type, note, omnibus_flag=False):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        INSERT INTO albums (catalog_no, title, artist, year, release_date, omnibus_flag, album_type, note)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    """, (catalog_no, title, artist, year, release_date, omnibus_flag, album_type, note))
    conn.commit()
    conn.close()

def get_albums():
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("SELECT id, catalog_no, title, artist, year, release_date, omnibus_flag, album_type, note FROM albums")
    rows = c.fetchall()
    conn.close()
    return rows

def get_album(album_id):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        SELECT id, catalog_no, title, artist, year, release_date, omnibus_flag, album_type, note
        FROM albums
        WHERE id = ?
    """, (album_id,))
    row = c.fetchone()
    conn.close()
    return row

def update_album(album_id, catalog_no, title, artist, year, release_date, album_type, note, omnibus_flag):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        UPDATE albums
        SET catalog_no=?, title=?, artist=?, year=?, release_date=?, album_type=?, note=?, omnibus_flag=?
        WHERE id=?
    """, (catalog_no, title, artist, year, release_date, album_type, note, omnibus_flag, album_id))
    conn.commit()
    conn.close()

# -----------------------------
# DB 操作（曲）
# -----------------------------
def add_track(album_id, track_no, title, artist, composer, arranger, lyricist, duration, note):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        INSERT INTO tracks (album_id, track_no, title, artist, composer, arranger, lyricist, duration, note)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
    """, (album_id, track_no, title, artist, composer, arranger, lyricist, duration, note))
    conn.commit()
    conn.close()

def update_track(track_id, track_no, title, artist, composer, arranger, lyricist, duration, note):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        UPDATE tracks
        SET track_no=?, title=?, artist=?, composer=?, arranger=?, lyricist=?, duration=?, note=?
        WHERE id=?
    """, (track_no, title, artist, composer, arranger, lyricist, duration, note, track_id))
    conn.commit()
    conn.close()

def get_tracks_by_album(album_id):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        SELECT id, track_no, title, duration, lyricist, composer, arranger, artist, note
        FROM tracks
        WHERE album_id = ?
        ORDER BY track_no ASC
    """, (album_id,))
    rows = c.fetchall()
    conn.close()
    return rows

def get_track(track_id):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        SELECT id, album_id, track_no, title, artist, composer, arranger, lyricist, duration, note
        FROM tracks
        WHERE id = ?
    """, (track_id,))
    row = c.fetchone()
    conn.close()
    return row

# -----------------------------
# DB 操作（ミュージシャン）
# -----------------------------
def add_musician(name, note):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("INSERT INTO musicians (name, note) VALUES (?, ?)", (name, note))
    conn.commit()
    conn.close()

def get_musicians():
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("SELECT id, name, note FROM musicians")
    rows = c.fetchall()
    conn.close()
    return rows

# -----------------------------
# DB 操作（曲 × ミュージシャン）
# -----------------------------
def add_track_musician(track_id, musician_id, instrument, note):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        INSERT INTO track_musicians (track_id, musician_id, instrument, note)
        VALUES (?, ?, ?, ?)
    """, (track_id, musician_id, instrument, note))
    conn.commit()
    conn.close()

def get_track_musicians(track_id):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        SELECT tm.id, m.name, tm.instrument, tm.note
        FROM track_musicians tm
        LEFT JOIN musicians m ON tm.musician_id = m.id
        WHERE tm.track_id = ?
    """, (track_id,))
    rows = c.fetchall()
    conn.close()
    return rows

# -----------------------------
# DB 操作（アルバム × ミュージシャン）
# -----------------------------
def add_album_musician(album_id, musician_id, instrument, note):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        INSERT INTO album_musicians (album_id, musician_id, instrument, note)
        VALUES (?, ?, ?, ?)
    """, (album_id, musician_id, instrument, note))
    conn.commit()
    conn.close()

def get_album_musicians(album_id):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        SELECT am.id, m.name, am.instrument, am.note, am.musician_id
        FROM album_musicians am
        LEFT JOIN musicians m ON am.musician_id = m.id
        WHERE am.album_id = ?
    """, (album_id,))
    rows = c.fetchall()
    conn.close()
    return rows

def delete_album_musician(album_musician_id):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("DELETE FROM album_musicians WHERE id = ?", (album_musician_id,))
    conn.commit()
    conn.close()

# -----------------------------
# DB 操作（アーティスト）
# -----------------------------
def add_artist(name, note=""):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("""
        INSERT INTO artists (name, note)
        VALUES (?, ?)
    """, (name, note))
    conn.commit()
    conn.close()

def get_artists():
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("SELECT id, name, note FROM artists ORDER BY name ASC")
    rows = c.fetchall()
    conn.close()
    return rows

def get_artist_by_name(name):
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute("SELECT id, name, note FROM artists WHERE name = ?", (name,))
    row = c.fetchone()
    conn.close()
    return row

# Streamlit UI

st.title("🎵 CD / 曲データ管理アプリ")

init_db()

if "view" not in st.session_state:
    st.session_state.view = "album_list"
if "current_album_id" not in st.session_state:
    st.session_state.current_album_id = None
if "current_track_id" not in st.session_state:
    st.session_state.current_track_id = None

menu = st.sidebar.selectbox(
    "メニュー",
    [
        "アルバム編集",
        "ミュージシャン登録"
        "アーティスト一覧"
    ]
)

# -----------------------------
# ミュージシャン登録
# -----------------------------
if menu == "ミュージシャン登録":
    st.header("👤 ミュージシャン登録")

    name = st.text_input("名前")
    note = st.text_input("備考（所属バンドなど）")

    if st.button("登録"):
        if name:
            add_musician(name, note)
            st.success("登録しました")
        else:
            st.error("名前は必須です")

# -----------------------------
# アルバム編集（アルバム情報＋曲一覧）
# -----------------------------
elif menu == "アルバム編集":

    # -----------------------------

        album_type = st.selectbox("種別", album_type_options)

        min_date = dt.date(1960, 1, 1)

        release_date = st.date_input("発売日", min_value=min_date, max_value=max_date, value=None)
        release_date_str = release_date.strftime("%Y-%m-%d") if release_date else ""

            year = release_date.year
            st.number_input("発売年（release_dateの年を自動設定）", value=year, min_value=1960, max_value=dt.datetime.today().year, disabled=True)
        else:
            year = st.number_input("発売年", min_value=1960, max_value=dt.datetime.today().year, step=1)

        note = st.text_area("備考")

        if st.button("登録"):
            if title:
        
                # --- アーティスト自動登録（空でなければ登録） ---
                if artist:  # 空文字の場合は登録しない

                st.session_state.view = "album_list"

            else:

    elif st.session_state.view == "album_edit":
        st.header("📀 アルバム編集")

        # アルバム選択
        albums = get_albums()
        album_dict = {f"{a[2]}（{a[3]}）": a[0] for a in albums}

        if len(album_dict) == 0:
            st.warning("アルバムがありません")
        else:
            selected_label = st.selectbox("編集するアルバム", list(album_dict.keys()))
            st.session_state.current_album_id = album_dict[selected_label]
            album = get_album(st.session_state.current_album_id)

            # 上部：アルバム情報表示・編集
            st.subheader("アルバム情報")

            title = st.text_input("アルバム名", value=album[2] or "")

            # --- アーティスト選択 UI（既存＋新規） ---
            artists = get_artists()  # [(id, name, note), ...]
            artist_names = [a[1] for a in artists]

            max_date = dt.datetime.today().date()

                min_value=min_date,

                current_artist = track[4] or ""
            else:
                current_artist = album[3] if album[6] == 0 else ""

                "アーティスト（既存）",
                ["新規入力"] + artist_names,

                artist = st.text_input("新規アーティスト名", value=current_artist)
